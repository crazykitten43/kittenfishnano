cmake_minimum_required(VERSION 3.14)
project(KittenFishMicro LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(micro src/board.cpp src/eval.cpp src/search.cpp)
target_include_directories(micro PUBLIC include)
add_executable(kittenfishmicro benchmarks/driver.cpp)
target_link_libraries(kittenfishmicro PRIVATE micro)
add_executable(micro_tests tests/tests.cpp)
target_link_libraries(micro_tests PRIVATE micro)
enable_testing()
add_test(NAME correctness COMMAND micro_tests)
